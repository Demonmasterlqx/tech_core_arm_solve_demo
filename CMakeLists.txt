cmake_minimum_required(VERSION 3.20)
project(technology_core VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TECH_CORE_VIEWER "Build GLFW interactive viewer and PNG capture" ON)
set(MUJOCO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.deps/mujoco-3.3.7" CACHE PATH "MuJoCo installation")
find_path(MUJOCO_INCLUDE_DIR mujoco/mujoco.h HINTS "${MUJOCO_ROOT}/include" "$ENV{MUJOCO_ROOT}/include" REQUIRED)
find_library(MUJOCO_LIBRARY NAMES mujoco HINTS "${MUJOCO_ROOT}/lib" "$ENV{MUJOCO_ROOT}/lib" REQUIRED)
add_library(mujoco SHARED IMPORTED)
set_target_properties(mujoco PROPERTIES IMPORTED_LOCATION "${MUJOCO_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${MUJOCO_INCLUDE_DIR}")
add_library(tech_core_rules src/rules.cpp)
target_include_directories(tech_core_rules PUBLIC include)
add_library(tech_core_sim src/simulation.cpp)
target_link_libraries(tech_core_sim PUBLIC tech_core_rules mujoco)
target_compile_definitions(tech_core_sim PUBLIC TECH_CORE_MODEL_PATH="${CMAKE_CURRENT_SOURCE_DIR}/models/tech_core.xml")
add_executable(tech_core src/main.cpp)
target_link_libraries(tech_core PRIVATE tech_core_sim)
if(TECH_CORE_VIEWER)
  find_package(glfw3 3.3 REQUIRED)
  find_package(OpenGL REQUIRED)
  find_package(PNG REQUIRED)
  target_sources(tech_core PRIVATE src/viewer.cpp)
  target_link_libraries(tech_core PRIVATE glfw OpenGL::GL PNG::PNG)
  target_compile_definitions(tech_core PRIVATE TECH_CORE_VIEWER=1)
endif()
foreach(target tech_core_rules tech_core_sim tech_core)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(rules_test tests/rules_test.cpp)
  target_link_libraries(rules_test PRIVATE tech_core_rules)
  add_test(NAME rules COMMAND rules_test)
  add_executable(physics_test tests/physics_test.cpp)
  target_link_libraries(physics_test PRIVATE tech_core_sim)
  add_test(NAME physics COMMAND physics_test)
  add_executable(geometry_test tests/geometry_test.cpp)
  target_link_libraries(geometry_test PRIVATE tech_core_sim)
  target_compile_definitions(geometry_test PRIVATE TECH_CORE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_options(geometry_test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME geometry_fidelity COMMAND geometry_test)
  foreach(level RANGE 1 4)
    add_test(NAME demo_level_${level} COMMAND tech_core --headless --demo --level ${level} --duration 15)
  endforeach()
  add_test(NAME demo_q_positive_limit COMMAND tech_core --headless --demo --level 3 --target 90 --duration 15)
  add_test(NAME demo_q_negative_limit COMMAND tech_core --headless --demo --level 3 --target -90 --duration 15)
  add_test(NAME demo_tilted_station COMMAND tech_core --headless --demo --level 3 --pose -0.1,0.25,0.65,25,75,-15 --duration 15)
  add_test(NAME cli_invalid_level COMMAND tech_core --headless --level 5)
  set_tests_properties(cli_invalid_level PROPERTIES WILL_FAIL TRUE)
endif()

include(cmake/rm26_arm.cmake)
include(cmake/planning.cmake)
include(cmake/planning_demo.cmake)
